# Human takeaway:
#   - Full-table analytics over ~50k bookings
#   - Hash Join properties <-> bookings
#   - WindowAgg for 7-day moving average + DENSE_RANK
#   - Seq Scan on bookings is normal here (query touches almost every row)
#   - Execution Time: 119.281 ms

EXPLAIN (ANALYZE, BUFFERS)
WITH daily_property_revenue AS (
    SELECT
        p.id AS property_id,
        p.title AS property_title,
        DATE_TRUNC('day', b.created_at)::DATE AS booking_date,
        SUM(b.total_cost) AS daily_revenue
    FROM properties p
    JOIN bookings b ON p.id = b.property_id
    GROUP BY p.id, p.title, DATE_TRUNC('day', b.created_at)::DATE
),
moving_average_calc AS (
    SELECT
        property_id,
        property_title,
        booking_date,
        daily_revenue,
        AVG(daily_revenue) OVER (
            PARTITION BY property_id
            ORDER BY booking_date
            ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
        ) AS moving_avg_7day
    FROM daily_property_revenue
)
SELECT
    property_id,
    property_title,
    booking_date,
    daily_revenue,
    ROUND(moving_avg_7day, 2) AS moving_avg_7day,
    DENSE_RANK() OVER (
        PARTITION BY booking_date
        ORDER BY moving_avg_7day DESC
    ) AS venue_rank
FROM moving_average_calc
ORDER BY booking_date DESC, venue_rank ASC;

# QUICK CHEAT SHEET — reading a plan line
# =============================================================================
#   Index Scan using idx_name     -> walked a B-tree index (good for selective queries)
#   Bitmap Index Scan on idx_name -> used index to build a bitmap of matching pages
#   Seq Scan on table             -> read whole table (OK if almost all rows needed)
#   Hash Join                     -> join via in-memory hash table
#   WindowAgg                     -> window function (AVG OVER / DENSE_RANK)
#   actual time=A..B              -> real start..end time in ms for that node
#   Execution Time: X ms          -> total wall time for the whole query
#
